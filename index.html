Can you refactor the auth callback to use a single async handler and add error handling?
Here's a consolidated handler with typed errors and a fallback for missing sessions: 
async function handleAuthCallback(req: Request): Promise<Response> {
  try {
    const session = await auth.callback(req);
    if (!session) throw new AuthError("no_session");
    return Response.redirect("/workspace");
  } catch (err) {
    return jsonError(err, "/login");
  }
}
